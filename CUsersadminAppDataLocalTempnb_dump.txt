# ── 2. DATASET DEFINITION ────────────────────────────────────────
class SkinDataset(Dataset):
    def __init__(self, df, transform=None):
        self.df = df.reset_index(drop=True)
        self.transform = transform
    def __len__(self):
        return len(self.df)
    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = path_map[row['image_id']]
        image = Image.open(img_path).convert('RGB')
        label = row['label_idx']
        if self.transform:
            image = self.transform(image)
        return image, label

# ── 3. CLASS BALANCING (SAMPLER) ─────────────────────────────────
def get_weighted_sampler(df):
    labels = df['label_idx'].values if 'label_idx' in df.columns else df.iloc[:, -1].values
    class_counts = np.bincount(labels)
    weights = 1. / class_counts
    samples_weights = torch.from_numpy(weights[labels])
    return WeightedRandomSampler(samples_weights, len(samples_weights), replacement=True)

# ── 4. TRANSFORMS & DATALOADERS ──────────────────────────────────
# ImageNet normalization — matches pretrained backbones (B0/B1/B2/R50/MNv2 all ImageNet)

    transforms.ColorJitter(brightness=0.2, contrast=0.2, saturation=0.2),
    transforms.ToTensor(),
    transforms.Normalize(norm_mean, norm_std)
])

val_tf = transforms.Compose([

class HierarchicalSkinDataset(Dataset):
    def __init__(self, df, transform=None, label_col='label_idx'):
        self.df = df.reset_index(drop=True)
        self.transform = transform
        self.label_col = label_col
    def __len__(self):
        return len(self.df)
    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = path_map[row['image_id']]
        image = Image.open(img_path).convert('RGB')
        label = row[self.label_col]
        if self.transform:
            image = self.transform(image)
        return image, label

def _weighted_sampler_for_col(df, label_col):
    labels = df[label_col].values
    class_counts = np.bincount(labels)
    weights = 1. / class_counts
    samples_weights = torch.from_numpy(weights[labels])
    return WeightedRandomSampler(samples_weights, len(samples_weights), replacement=True)

def create_dataloaders(df, label_col, test_size=0.15, val_size_frac=0.1765, batch_size=BATCH_SIZE):
    train_val_subset, test_subset = train_test_split(
        df, test_size=test_size, stratify=df[label_col], random_state=42
    )
    train_subset, val_subset = train_test_split(
        train_val_subset, test_size=val_size_frac, stratify=train_val_subset[label_col], random_state=42
    )

    train_ds = HierarchicalSkinDataset(train_subset, train_tf, label_col=label_col)
    val_ds   = HierarchicalSkinDataset(val_subset,   val_tf,   label_col=label_col)
    test_ds  = HierarchicalSkinDataset(test_subset,  val_tf,   label_col=label_col)

    if len(df[label_col].unique()) > 1:
        train_sampler = _weighted_sampler_for_col(train_subset, label_col)
        train_loader = DataLoader(train_ds, batch_size=batch_size, sampler=train_sampler, num_workers=2, pin_memory=True)
    else:
        train_loader = DataLoader(train_ds, batch_size=batch_size, shuffle=True, num_workers=2, pin_memory=True)

    val_loader  = DataLoader(val_ds,  batch_size=batch_size, shuffle=False, num_workers=2, pin_memory=True)
    test_loader = DataLoader(test_ds, batch_size=batch_size, shuffle=False, num_workers=2, pin_memory=True)
    return train_loader, val_loader, test_loader

import torch.nn as nn
import torchvision.models as models

def create_model(num_classes, backbone=None, pretrained=True, dropout_rate=None):
    """
    Multi-backbone classifier head builder.
    backbone: one of 'effnet_b0', 'effnet_b1', 'effnet_b2', 'resnet50', 'mobilenetv2'.
              Defaults to global BACKBONE.
    dropout_rate: defaults to backbone-specific DROPOUT from CFG.
    """
    if backbone is None:
        backbone = BACKBONE
    if dropout_rate is None:
        dropout_rate = DROPOUT

    if backbone == 'effnet_b0':
        model = models.efficientnet_b0(pretrained=pretrained)
        in_f = model.classifier[1].in_features
        model.classifier = nn.Sequential(nn.Dropout(p=dropout_rate, inplace=True), nn.Linear(in_f, num_classes))

    elif backbone == 'effnet_b1':
        model = models.efficientnet_b1(pretrained=pretrained)
        in_f = model.classifier[1].in_features
        model.classifier = nn.Sequential(nn.Dropout(p=dropout_rate, inplace=True), nn.Linear(in_f, num_classes))

    elif backbone == 'effnet_b2':
        model = models.efficientnet_b2(pretrained=pretrained)
        in_f = model.classifier[1].in_features
        model.classifier = nn.Sequential(nn.Dropout(p=dropout_rate, inplace=True), nn.Linear(in_f, num_classes))

    elif backbone == 'resnet50':
        model = models.resnet50(pretrained=pretrained)
        in_f = model.fc.in_features
        model.fc = nn.Sequential(nn.Dropout(p=dropout_rate, inplace=True), nn.Linear(in_f, num_classes))

    elif backbone == 'mobilenetv2':
        model = models.mobilenet_v2(pretrained=pretrained)
        in_f = model.classifier[1].in_features
        model.classifier = nn.Sequential(nn.Dropout(p=dropout_rate, inplace=True), nn.Linear(in_f, num_classes))

    else:
        raise ValueError(f"Unknown backbone: {backbone}")

    return model

 import torch.nn.functional as F

class FocalLoss(nn.Module):
    """
    Focal Loss, as described in https://arxiv.org/abs/1708.02002.
    It modifies the standard cross-entropy loss to focus learning on hard misclassified examples.
    """
    def __init__(self, alpha=1.0, gamma=2.0, class_weights=None):
        super(FocalLoss, self).__init__()
        self.alpha = alpha
        self.gamma = gamma
        self.class_weights = class_weights

    def forward(self, inputs, targets):
        weight = self.class_weights
        if weight is not None and weight.device != inputs.device:
            weight = weight.to(inputs.device)
        ce_loss = F.cross_entropy(inputs, targets, reduction='none', weight=weight)
        pt = torch.exp(-ce_loss)
        focal_loss = self.alpha * (1 - pt)**self.gamma * ce_loss
        return focal_loss.mean()

# Cell 5: Training and Evaluation Loop
# This cell defines the main function for training and validating a model.
from tqdm.notebook import tqdm
import time
import json
import os
from sklearn.metrics import balanced_accuracy_score, f1_score

def compute_class_weights(df, label_col, power=1.0):
    """
    Computes inverse-frequency class weights for the given label column.
    """
    counts = df[label_col].value_counts().sort_index()
    weights = 1.0 / (counts ** power)
    weights = weights / weights.sum() * len(weights)
    return torch.tensor(weights.values, dtype=torch.float)

def train_model(
    model, train_loader, val_loader, model_name, criterion,
    epochs=20, learning_rate=0.001, weight_decay=1e-4,
    scheduler_patience=2, scheduler_factor=0.5, monitor_metric="val_bal_acc", mix_weight=0.5
    ):
    """
    A complete function to train and evaluate a model.
    
    Args:
        model (nn.Module): The model to train.
        train_loader (DataLoader): DataLoader for the training set.
        val_loader (DataLoader): DataLoader for the validation set.
        model_name (str): A name for the model, used for saving the best weights.
        criterion: The loss function to use.
        epochs (int): Number of training epochs.
        learning_rate (float): Learning rate for the optimizer.
        weight_decay (float): Weight decay for AdamW.
        scheduler_patience (int): Patience for ReduceLROnPlateau.
        scheduler_factor (float): LR reduction factor for ReduceLROnPlateau.
        monitor_metric (str): Metric used for checkpointing and scheduler.
        mix_weight (float): Weight for val_acc in blended metric (val_mix).
        
    Returns:
        A dictionary containing training history (loss, accuracy, balanced accuracy, f1).
    """
    # Setup: optimizer and device
    optimizer = optim.AdamW(model.parameters(), lr=learning_rate, weight_decay=weight_decay)
    scheduler = optim.lr_scheduler.ReduceLROnPlateau(
        optimizer, mode="max", factor=scheduler_factor, patience=scheduler_patience
    )
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model.to(device)
    if hasattr(criterion, "weight") and criterion.weight is not None:
        criterion.weight = criterion.weight.to(device)
    
    # Create history directory
    history_dir = "History"
    os.makedirs(history_dir, exist_ok=True)
    
    print(f"🚀 Starting training for {model_name} on {device}...")
    
    best_val_metric = 0.0
    history = {
        'train_loss': [], 'train_acc': [], 'val_loss': [], 'val_acc': [],
        'train_bal_acc': [], 'val_bal_acc': [], 'train_f1': [], 'val_f1': []
    }

    for epoch in range(epochs):
        start_time = time.time()
        
        # --- Training Phase ---
        model.train()
        running_loss = 0.0
        correct_predictions = 0
        total_samples = 0
        train_preds, train_labels = [], []
        
        for images, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}/{epochs} [Train]"):
            images, labels = images.to(device), labels.to(device)
            
            optimizer.zero_grad()
            outputs = model(images)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
            
            running_loss += loss.item() * images.size(0)
            _, predicted = torch.max(outputs.data, 1)
            total_samples += labels.size(0)
            correct_predictions += (predicted == labels).sum().item()
            train_preds.extend(predicted.detach().cpu().numpy())
            train_labels.extend(labels.detach().cpu().numpy())
            
        epoch_train_loss = running_loss / total_samples
        epoch_train_acc = correct_predictions / total_samples
        epoch_train_bal_acc = balanced_accuracy_score(train_labels, train_preds)
        epoch_train_f1 = f1_score(train_labels, train_preds, average='macro')
        history['train_loss'].append(epoch_train_loss)
        history['train_acc'].append(epoch_train_acc)
        history['train_bal_acc'].append(epoch_train_bal_acc)
        history['train_f1'].append(epoch_train_f1)

        # --- Validation Phase ---
        model.eval()
        running_loss = 0.0
        correct_predictions = 0
        total_samples = 0
        val_preds, val_labels = [], []
        
        with torch.no_grad():
            for images, labels in tqdm(val_loader, desc=f"Epoch {epoch+1}/{epochs} [Val]"):
                images, labels = images.to(device), labels.to(device)
                outputs = model(images)
                loss = criterion(outputs, labels)
                
                running_loss += loss.item() * images.size(0)
                _, predicted = torch.max(outputs.data, 1)
                total_samples += labels.size(0)
                correct_predictions += (predicted == labels).sum().item()
                val_preds.extend(predicted.detach().cpu().numpy())
                val_labels.extend(labels.detach().cpu().numpy())

        epoch_val_loss = running_loss / total_samples
        epoch_val_acc = correct_predictions / total_samples
        epoch_val_bal_acc = balanced_accuracy_score(val_labels, val_preds)
        epoch_val_f1 = f1_score(val_labels, val_preds, average='macro')
        history['val_loss'].append(epoch_val_loss)
        history['val_acc'].append(epoch_val_acc)
        history['val_bal_acc'].append(epoch_val_bal_acc)
        history['val_f1'].append(epoch_val_f1)
        
        # --- Scheduler, Log, and Save Best Model ---
        if monitor_metric == "val_acc":
            monitor_value = epoch_val_acc
        elif monitor_metric == "val_f1":
            monitor_value = epoch_val_f1
        elif monitor_metric == "val_mix":
            monitor_value = (mix_weight * epoch_val_acc) + ((1 - mix_weight) * epoch_val_bal_acc)
        else:
            monitor_value = epoch_val_bal_acc
        scheduler.step(monitor_value)
        elapsed_time = time.time() - start_time
        current_lr = optimizer.param_groups[0]['lr']
        print(
            f"Epoch {epoch+1}/{epochs} - {elapsed_time:.2f}s - LR: {current_lr:.2e} - "
            f"Train Loss: {epoch_train_loss:.4f}, Train Acc: {epoch_train_acc:.4f}, "
            f"Train Bal Acc: {epoch_train_bal_acc:.4f}, Train F1: {epoch_train_f1:.4f} | "
            f"Val Loss: {epoch_val_loss:.4f}, Val Acc: {epoch_val_acc:.4f}, "
            f"Val Bal Acc: {epoch_val_bal_acc:.4f}, Val F1: {epoch_val_f1:.4f}"
        )

        if monitor_value > best_val_metric:
            best_val_metric = monitor_value
            # Save the model's state dictionary
            torch.save(model.state_dict(), f'{model_name}_best.pth')
            print(f"✨ New best model saved with {monitor_metric}: {best_val_metric:.4f}")
            
    print(f"🏁 Finished training for {model_name}.")
    
    # Save history to a JSON file
    history_path = os.path.join(history_dir, f'{model_name}_history.json')
    with open(history_path, 'w') as f:
        json.dump(history, f)
    print(f"💾 Training history saved to {history_path}")
    
    return history

    def __init__(self, mean, std):
        self.mean, self.std = mean, std
    def __call__(self, tensor):
        t = tensor.clone()
        for t_i, m, s in zip(t, self.mean, self.std):

        return t

def tta_forward(model, images, n=4):
    """Test-time aug: original + h-flip + v-flip + hv-flip. Avg logits."""
    with torch.no_grad():
        out  = model(images)
        out += model(torch.flip(images, dims=[3]))
        out += model(torch.flip(images, dims=[2]))
        out += model(torch.flip(images, dims=[2, 3]))
    return out / n

def get_all_preds_and_labels(model, dataloader, use_tta=True):
    model.to(DEVICE)
    all_preds, all_labels = [], []
    with torch.no_grad():
        for images, labels in dataloader:
            images = images.to(DEVICE)
            outputs = tta_forward(model, images) if use_tta else model(images)
            preds = torch.argmax(outputs, dim=1)
            all_preds.extend(preds.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
    model.to('cpu')
    return np.array(all_preds), np.array(all_labels)

    return None

    else:
        print(f"⚠️ Warning: Image ID {img_id} not found in path_map.")

print(f"\n✅ Export Complete!")
print(f"📊 Total images copied: {copy_count}")
print(f"📂 All files are located in: {export_dir}")
